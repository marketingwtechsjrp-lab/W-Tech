# robots.txt controla RASTREAMENTO, não indexação. Para tirar uma página do índice
# use <meta name="robots" content="noindex"> e mantenha o rastreamento liberado —
# um Disallow aqui impede o robô de LER o noindex.
#
# Nunca bloquear /assets/ nem os módulos JS: sem eles o crawler não consegue
# renderizar a página e vê um site quebrado.

User-agent: *
Allow: /

# Painel e API
Disallow: /admin
Disallow: /api/
Disallow: /auth/

# Áreas privadas do cliente
Disallow: /meus-pedidos

# Fluxo transacional — sem valor de busca e gera duplicata
Disallow: /checkout
Disallow: /checkout-curso
Disallow: /checkout-lisboa
Disallow: /pagamento-sucesso
Disallow: /inscricao-confirmada
Disallow: /obrigado-lisboa
Disallow: /obrigado-suspensao

# Campanhas de captura: variações quase idênticas da mesma oferta
Disallow: /captura/

# Parâmetros de rastreamento: LIBERADOS de propósito.
#
# Bloqueá-los saiu pela culatra. O Search Console lista 7 páginas "indexadas,
# mas bloqueadas pelo robots.txt", e todas são páginas reais decoradas pelo
# GTM — /mapa, /cursos, /blog, /molas, /privacidade e /cancelamento com
# ?utm_source=direto&sck=<id da sessão> na ponta.
#
# O mecanismo é o descrito no topo deste arquivo: cada uma dessas URLs já traz
# um <link rel="canonical"> apontando para a versão limpa, que consolidaria as
# duas. Só que o Disallow impedia o robô de BUSCAR a página — e sem buscar, ele
# não lê o canonical. Resultado: a URL com parâmetro entra no índice sozinha,
# sem conteúdo, competindo com a página de verdade.
#
# Com o rastreamento liberado, o canonical faz o trabalho e as versões com
# parâmetro somem do índice.
